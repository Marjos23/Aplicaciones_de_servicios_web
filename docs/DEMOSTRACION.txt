############ 1) MATRÍCULA VÁLIDA ############

============================================================
# POST /enrollments (Ana=1 en curso 2) -> 201
============================================================
-> -X POST http://localhost:3000/enrollments -H Content-Type: application/json -d {"studentId":1,"courseId":2}
HTTP 201
{
  "id": 2,
  "studentId": 1,
  "courseId": 2
}

############ 2) MATRÍCULA DUPLICADA ############

============================================================
# POST /enrollments repetido (Ana=1, curso=2) -> 409
============================================================
-> -X POST http://localhost:3000/enrollments -H Content-Type: application/json -d {"studentId":1,"courseId":2}
HTTP 409
{
  "message": "El estudiante 1 ya está matriculado en el curso 2",
  "error": "Conflict",
  "statusCode": 409
}
